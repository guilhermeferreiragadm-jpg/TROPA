> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Registro de peso e medidas com tendência, fotos de progresso em miniatura, faixa de calorias só para adultos e um cartão semanal de pesagem.

**Architecture:** Regras novas viram funções puras no bloco `/* @pure:start */ ... /* @pure:end */` de `publicar.html`, testadas em Node. A interface é uma tela nova "Corpo" (subtela do Perfil) mais um cartão na tela Hoje e uma linha na tela Comida, reaproveitando `tela`, `topo`, `miniatura`, `salvarLog`, `salvarPerfil`, `bindGo`.

**Tech Stack:** HTML e JS puro em um arquivo (`publicar.html`), `node --test tests/*.test.js` no Node 24, artifact do claude.ai (`window.claude`).

**Spec:** `docs/superpowers/specs/2026-10-27-fase3-corpo-design.md`

## Global Constraints

- Um único arquivo de app: `publicar.html`. Não criar outros arquivos de código.
- Comando de teste: `node --test tests/*.test.js` (no Node 24 passar só a pasta falha).
- Texto de interface em português do Brasil, sem travessão no lugar de vírgula.
- O app só roda dentro do claude.ai: nada de `localStorage` para dados importantes; dados novos vão em `LOG` via `salvarLog(chave, valor)`. `LOG` tem limite de 256 KB: guardar só texto curto e miniaturas pequenas.
- Menores de 18 anos (`P.perfil.menor`): só registram peso. Sem medidas, sem foto, sem tendência, sem faixa de calorias, sem lembrete.
- Nunca meta de peso, nunca texto que comemore perda rápida. Foto, medidas e peso nunca vão para o feed, para `social` nem para a IA como imagem.
- O carregador de `tests/pure.test.js` devolve uma lista de nomes (`return {...}`): ao incluir uma função pura nova, acrescente o nome no fim da lista e mantenha todos os anteriores.
- Funções puras novas entram no bloco `@pure`, imediatamente antes da linha `const fotoValida = s =>` (confira com `grep -c "^const fotoValida = s =>" publicar.html` = 1). As funções puras desta fase não dependem de nenhum outro helper (nem `dow`, nem `key`): calculam datas localmente.
- Cada task termina com commit na branch `fase3-corpo-dados`. Mensagem de commit termina com `Co-Authored-By: Claude Sonnet 5.5 <noreply@anthropic.com>`.
- Arquivo tem linhas muito longas: use Edit com trechos curtos e únicos; confira com `grep -c` que o trecho aparece uma vez antes de editar. Se um trecho âncora não existir exatamente como descrito, pare e reporte.
- Convenções do app já existentes: `$(id)`, `esc(s)`, `num(v, c)` (formata número com vírgula; `c` verdadeiro = 1 casa), `toast(t)`, `today` (meia-noite), `key(d)` (AAAA-MM-DD), `addDays`, `dow(d)` (segunda = 0), `fmt(d)`, `parseD(s)`, `tela(html, opts)` (devolve o elemento raiz), `topo(k, h1)`, `bindGo(root)` (botões `data-go="aba|sub|arg"`), `LOG`, `salvarLog(k, v)` (promessa), `salvarPerfil()`, `render(keep)`, `$("ov")` para sheets (`<div class="dim" id="dim"></div><div class="sheet" role="dialog">…`), `miniatura(file, lado, q)` (devolve data URL JPEG ou ""), `fotoValida(s)` (data URL JPEG de até 60000 caracteres), classes `.box`, `.btn`, `.btn.go`, `.btn.ghost`, `.btn.sm`, `.lk`, `.k`, `.s`, `.msg`, `.fld`, `.inp`, `.grid2`, `.between`, `.chips`, `.chip`.

```js
const CHAVES_MEDIDA = ["cin","qua","pei","bra","cox"];
function registrarCorpo(lista, reg, menor){
  const ok = s => /^\d{4}-\d{2}-\d{2}$/.test(String(s));
  const l = Array.isArray(lista) ? lista.filter(x=>x && ok(x.d)) : [];
  const kg = Math.round(+(reg && reg.kg)*10)/10;
  if (!reg || !ok(reg.d) || !(kg >= 30 && kg <= 300)) return l;
  const novo = {d:reg.d, kg};
  if (!menor){
    const m = {};
    CHAVES_MEDIDA.forEach(k=>{ const v = Math.round(+(reg.m && reg.m[k])*10)/10; if (v >= 10 && v <= 250) m[k] = v; });
    if (Object.keys(m).length) novo.m = m;
  }
  return l.filter(x=>x.d !== reg.d).concat([novo]).sort((a,b)=>a.d < b.d ? -1 : a.d > b.d ? 1 : 0).slice(-120);
}
function tendenciaPeso(lista, hoje){
  const dia = s => Date.UTC(+s.slice(0,4), +s.slice(5,7)-1, +s.slice(8,10))/864e5, h = dia(hoje);
  const pts = (Array.isArray(lista) ? lista : []).filter(x=>x && typeof x.d === "string" && +x.kg > 0).map(x=>({t:dia(x.d)-h, kg:+x.kg})).filter(p=>p.t >= -28 && p.t <= 0);
  if (pts.length < 3) return null;
  const ts = pts.map(p=>p.t), dias = Math.max(...ts) - Math.min(...ts);
  if (dias < 14) return null;
  const n = pts.length, mt = pts.reduce((a,p)=>a+p.t,0)/n, mk = pts.reduce((a,p)=>a+p.kg,0)/n;
  const sxx = pts.reduce((a,p)=>a+(p.t-mt)*(p.t-mt),0), sxy = pts.reduce((a,p)=>a+(p.t-mt)*(p.kg-mk),0);
  const atual = pts.reduce((a,p)=>p.t > a.t ? p : a);
  return {kgSemana:Math.round(sxy/sxx*7*10)/10 || 0, pesoAtual:atual.kg, n, dias};
}
function variacaoMedidas(lista){
  const out = {}, l = Array.isArray(lista) ? lista : [];
  CHAVES_MEDIDA.forEach(k=>{
    const v = l.filter(x=>x && x.m && +x.m[k] > 0);
    if (v.length >= 2){ const a = v[0], b = v[v.length-1]; out[k] = {de:a.m[k], para:b.m[k], dif:Math.round((b.m[k]-a.m[k])*10)/10, desde:a.d}; }
  });
  return out;
}
function lembretePesagem(lista, cfg, hoje, dispensado, menor){
  if (menor || dispensado || !cfg || !cfg.lembrar || !Array.isArray(lista) || !lista.length) return false;
  const dia = s => Date.UTC(+s.slice(0,4), +s.slice(5,7)-1, +s.slice(8,10))/864e5, h = dia(hoje);
  const sem = (h + 3) % 7, seg = h - sem;   // 1970-01-01 foi quinta: (dias + 3) % 7 dá segunda = 0
  if (sem < (+cfg.dia || 0)) return false;
  return !lista.some(x=>x && typeof x.d === "string" && dia(x.d) >= seg);
}
```

```js
function faixaKcal(c){
  c = c || {};
  const kg = +c.kg, cm = +c.cm, id = +c.idade;
  if (c.menor || (c.red && c.red.length) || c.ciclo === "sim" || !(kg > 0) || !(cm > 0) || !(id > 0)) return null;
  const bmr = 10*kg + 6.25*cm - 5*id + (c.sexo === "f" ? -161 : c.sexo === "m" ? 5 : -78);
  const manut = bmr*1.35 + (+c.treinoDia > 0 ? +c.treinoDia : 0);
  let lo = 0.95, hi = 1.05, motivo = "Para manter o peso e treinar bem, perto do que você gasta";
  if (c.objetivo === "emagrecer"){ lo = 0.80; hi = 0.90; motivo = "Para emagrecer com saúde, um pouco abaixo do que você gasta"; }
  else if (c.objetivo === "massa"){ lo = 1.05; hi = 1.10; motivo = "Para ganhar massa, um pouco acima do que você gasta"; }
  let ajuste = 0;
  const t = c.tendenciaKgSemana, pct = (t === null || t === undefined || !isFinite(+t)) ? null : +t/kg*100, longa = +c.diasTendencia >= 21;
  if (pct !== null){
    if (c.objetivo === "emagrecer"){
      if (pct <= -0.9){ ajuste = 150; motivo = "Você estava perdendo peso rápido demais, então a faixa subiu um pouco"; }
      else if (pct > -0.1 && longa) motivo += ". O peso parou: converse com o coach antes de mudar a comida";
    } else if (c.objetivo === "massa"){
      if (pct >= 0.5){ ajuste = -100; motivo = "Você estava ganhando peso rápido, então a faixa desceu um pouco"; }
      else if (pct < 0.1 && longa){ ajuste = 150; motivo = "O peso não subiu em 3 semanas, então a faixa subiu um pouco"; }
    }
  }
  const piso = Math.max(bmr, c.sexo === "f" ? 1200 : c.sexo === "m" ? 1500 : 1350);
  const r50 = v => Math.round(v/50)*50;
  const min = r50(Math.max(manut*lo + ajuste, piso));
  const max = r50(Math.max(manut*hi + ajuste, min + 100));
  return {min, max, manutencao:r50(manut), motivo};
}
```
